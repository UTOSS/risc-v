% ========================================================
% INTERCONNECT WIRING
% ========================================================

\def\hop{0.22}

% 1. "instruction" input to csr_decode
% (driven from decode_stage internals, so the wire starts inside the stage)
\draw[thickline] (-0.60, 0.60) -- (csr_decode-instruction);

% 2. "csr_request": out of csr_decode to a split point; one arm drops into
%    csr_data, the other rises over csr_data and runs through the pipeline,
%    jogging down into csr_wb row 1
\draw[thickline] (csr_decode-csr_request) -- (5.70, 0.60) |- (6.10, -0.20);
\draw[thickline] (5.70, 0.60) -- (5.70, 2.05) -- (28.25, 2.05) |- (csr_wb-csr_request);
\fill[arrowgreen] (5.70, 0.60) circle (3.2pt);

% 3. read address: only `address` of csr_request is used within csr_data; it
%    fans out to every plugin's read_address, hopping over the csr_wb_request
%    bus, and continues (dashed) to any further plugins
\draw[thinline] (6.10, -0.20) -- (pluginA-read_address);
\draw[thinline]
  (7.45, -0.20) -- (7.45, -3.60)
  -- (7.90 - \hop, -3.60) arc[start angle=180, end angle=0, radius=\hop]
  -- (pluginB-read_address);
\draw[thinline, dashed] (7.45, -3.60) -- (7.45, -5.75);
\fill[arrowgreen] (7.45, -0.20) circle (2.4pt);
\fill[arrowgreen] (7.45, -3.60) circle (2.4pt);
\node[wirelabel, anchor=south west] at (6.16, -0.14) {address};

% 4. read_value of every plugin OR-ed into "csr_data", which runs straight
%    through the pipeline into csr_wb row 2; further plugins join in dashed
\draw[thickline] (pluginA-read_value) -| ($(readvalueor.input 1) + (-0.80, 0)$) -- (readvalueor.input 1);
\draw[thickline] (pluginB-read_value)  -| ($(readvalueor.input 2) + (-0.80, 0)$) -- (readvalueor.input 2);
\draw[thickline, dashed] (11.25, -5.75) -| ($(readvalueor.input 3) + (-0.40, 0)$) -- (readvalueor.input 3);
\draw[thickline] (readvalueor.output) -- ++(0.60, 0) |- (csr_wb-csr_data);

% 5. "write_intent": thin taps pulled out of the csr_request bus after each
%    pipeline register (_e, _m, _w) and dropped into the hazard unit; each
%    hops (radius \hop) over the csr_data wire
\foreach \xd/\yt/\lbl in {21.35/2.05/e, 25.25/2.05/m, 28.25/0.60/w} {
  \draw[thinline]
    (\xd, \yt) -- (\xd, -0.20 + \hop) arc[start angle=90, end angle=-90, radius=\hop]
    -- (\xd, -11.20);
  \fill[arrowgreen] (\xd, \yt) circle (3.2pt);
  \node[wirelabel, anchor=west] at (\xd + 0.12, -10.35) {write\_intent\_\lbl};
}

% 6. "csr_wb_request" write-back bus: csr_wb right -> down below the pipeline
%    registers -> left -> up into csr_data from below (not pipelined), fanning
%    out to every plugin's csr_wb_request. Hops (radius \hop) over each
%    write_intent wire it crosses.
\draw[thickline]
  (csr_wb-csr_wb_request) -- (35.05, 0.60) -- (35.05, -9.50)
  \foreach \xc in {28.25, 25.25, 21.35} {
    -- (\xc + \hop, -9.50) arc[start angle=0, end angle=180, radius=\hop]
  }
  -- (7.90, -9.50)
  |- (pluginA-csr_wb_request);
\draw[thickline] (7.90, -4.40) -- (pluginB-csr_wb_request);
\fill[arrowgreen] (7.90, -4.40) circle (3.2pt);
